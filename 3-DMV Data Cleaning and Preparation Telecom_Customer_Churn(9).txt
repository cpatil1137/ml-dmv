import pandas as pd, numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

df = pd.read_csv("Telecom_Customer_Churn.csv")
print(df.shape); print(df.info()); print(df.isna().sum())

df = df.drop_duplicates()
df["gender"] = df.gender.str.title()
df["TotalCharges"] = pd.to_numeric(df.TotalCharges, errors="coerce")

df["MonthlyCharges"] = df.MonthlyCharges.fillna(df.MonthlyCharges.median())
df["TotalCharges"] = df.TotalCharges.fillna(df.TotalCharges.median())

q1, q3 = df.MonthlyCharges.quantile([.25, .75])
df = df[df.MonthlyCharges.between(q1 - 1.5 * (q3 - q1), q3 + 1.5 * (q3 - q1))]

df["AvgChargePerMonth"] = df.TotalCharges / (df.tenure + 1)
df["IsLongTerm"] = (df.Contract != "Month-to-month").astype(int)
df["Churn"] = df.Churn.map({"Yes": 1, "No": 0})

df = pd.get_dummies(df.drop(columns="customerID"), drop_first=True)
num = ["tenure", "MonthlyCharges", "TotalCharges", "AvgChargePerMonth"]
df[num] = StandardScaler().fit_transform(df[num])

X, y = df.drop(columns="Churn"), df.Churn
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=1)
print("Train/test:", X_train.shape, X_test.shape)

df.to_csv("data/Telecom_Cleaned.csv", index=False)
print("Saved data/Telecom_Cleaned.csv")
